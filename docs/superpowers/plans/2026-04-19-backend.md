```typescript
import { readFile, writeFile, unlink, mkdir, readdir } from 'fs/promises'
import { join, dirname } from 'path'

export class FileStore {
  private baseDir: string

  constructor(baseDir: string) {
    this.baseDir = baseDir
  }

  private resolvePath(filename: string): string {
    return join(this.baseDir, filename)
  }

  async ensureDir(filename: string): Promise<void> {
    const dir = dirname(this.resolvePath(filename))
    await mkdir(dir, { recursive: true })
  }

  async read<T>(filename: string, defaultValue?: T): Promise<T | null> {
    try {
      const content = await readFile(this.resolvePath(filename), 'utf-8')
      return JSON.parse(content) as T
    } catch {
      return defaultValue !== undefined ? defaultValue : null
    }
  }

  async write<T>(filename: string, data: T): Promise<void> {
    await this.ensureDir(filename)
    const content = JSON.stringify(data, null, 2)
    await writeFile(this.resolvePath(filename), content, 'utf-8')
  }

  async delete(filename: string): Promise<void> {
    try {
      await unlink(this.resolvePath(filename))
    } catch {
      // File may not exist, ignore
    }
  }

  async listFiles(subdir: string): Promise<string[]> {
    try {
      const dir = subdir ? join(this.baseDir, subdir) : this.baseDir
      const entries = await readdir(dir, { withFileTypes: true })
      return entries.filter((e) => e.isFile() && e.name.endsWith('.json')).map((e) => e.name)
    } catch {
      return []
    }
  }

  async init(): Promise<void> {
    await mkdir(this.baseDir, { recursive: true })
    await mkdir(join(this.baseDir, 'conversations'), { recursive: true })
  }
}
```

```typescript
import type { OllamaChatRequest, OllamaChatResponse, OllamaModel, OllamaTagsResponse } from '../storage/types.ts'

export class OllamaService {
  private baseUrl: string

  constructor(baseUrl: string = 'http://localhost:11434') {
    this.baseUrl = baseUrl
  }

  async listModels(): Promise<OllamaModel[]> {
    const res = await fetch(`${this.baseUrl}/api/tags`)
    if (!res.ok) throw new Error(`Ollama API error: ${res.status}`)
    const data: OllamaTagsResponse = await res.json()
    return data.models
  }

  async chat(request: OllamaChatRequest): Promise<OllamaChatResponse> {
    const res = await fetch(`${this.baseUrl}/api/chat`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ...request, stream: false }),
    })
    if (!res.ok) throw new Error(`Ollama API error: ${res.status}`)
    return res.json()
  }

  async chatStream(request: OllamaChatRequest): Promise<ReadableStream<Uint8Array>> {
    const res = await fetch(`${this.baseUrl}/api/chat`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ...request, stream: true }),
    })
    if (!res.ok) throw new Error(`Ollama API error: ${res.status}`)
    if (!res.body) throw new Error('No response body for streaming')
    return res.body
  }

  async isHealthy(): Promise<boolean> {
    try {
      const res = await fetch(`${this.baseUrl}/`)
      return res.ok
    } catch {
      return false
    }
  }
}
```

```typescript
import { SYSTEM_PROMPT } from '../systemPrompt.ts'
import type { Message, MemoryEntry, Settings } from '../storage/types.ts'

export class ContextService {
  assemble(
    messages: Message[],
    newMessage: string,
    settings: Settings,
    memories: MemoryEntry[],
    summary: string,
  ): { role: string; content: string }[] {
    const context: { role: string; content: string }[] = []

    // Layer 1: System prompt (always present)
    context.push({ role: 'system', content: SYSTEM_PROMPT })

    // Layer 2: User preferences (if set)
    if (settings.preferencesSet) {
      const prefs = settings.preferences
      const prefLines: string[] = ['[User Preferences]']
      if (prefs.responseDetail) prefLines.push(`Detail level: ${prefs.responseDetail}`)
      if (prefs.technicalLevel) prefLines.push(`Technical level: ${prefs.technicalLevel}`)
      if (prefs.responseFormat) prefLines.push(`Response format: ${prefs.responseFormat}`)
      if (prefs.domains) prefLines.push(`Domains: ${prefs.domains}`)
      if (prefs.personality) prefLines.push(`Personality: ${prefs.personality}`)
      if (prefs.proactiveness) prefLines.push(`Proactiveness: ${prefs.proactiveness}`)
      if (prefs.languageStyle) prefLines.push(`Language: ${prefs.languageStyle}`)
      if (prefs.useOfExamples) prefLines.push(`Examples: ${prefs.useOfExamples}`)
      if (prefLines.length > 1) {
        context.push({ role: 'system', content: prefLines.join('\n') })
      }
    }

    // Layer 3: Persistent memory (active entries only)
    const activeMemories = memories.filter((m) => m.active)
    if (activeMemories.length > 0) {
      const memLines = ['[Remembered Facts]', ...activeMemories.map((m) => `- ${m.fact}`)]
      context.push({ role: 'system', content: memLines.join('\n') })
    }

    // Layer 4: Conversation summary
    if (summary) {
      context.push({ role: 'system', content: `[Earlier Conversation Summary]\n${summary}` })
    }

    // Layer 5: Rolling window
    const windowSize = settings.contextWindowSize
    const windowMessages = messages.slice(-windowSize)
    for (const msg of windowMessages) {
      context.push({ role: msg.role, content: msg.content })
    }

    // Layer 6: New user message
    context.push({ role: 'user', content: newMessage })

    return context
  }
}
```

```typescript
      messages: [],

    const conversations: Pick<Conversation, 'id' | 'title' | 'model' | 'updatedAt'>[] = []
    for (const file of files) {
      const conv = await this.store.read<Conversation>(`conversations/${file}`)
      if (conv) {
        conversations.push({ id: conv.id, title: conv.title, model: conv.model, updatedAt: conv.updatedAt })
      }
    }
    return conversations.sort((a, b) => b.updatedAt - a.updatedAt)
  }

    return this.store.read<Conversation>(`conversations/${id}.json`)
  }

    const conv = await this.get(id)
    if (!conv) throw new Error(`Conversation ${id} not found`)
    conv.messages.push(message)
    conv.updatedAt = Date.now()
    // Auto-title from first user message
    if (conv.title === 'New Conversation' && message.role === 'user') {
      conv.title = message.content.slice(0, 50)
    }
    await this.store.write(`conversations/${id}.json`, conv)
    return conv
```

```typescript
import { v4 as uuidv4 } from 'uuid'
import type { MemoryEntry } from '../storage/types.ts'
import type { FileStore } from '../storage/fileStore.ts'

export class MemoryService {
  private store: FileStore

  constructor(store: FileStore) {
    this.store = store
  }

  async list(): Promise<MemoryEntry[]> {
    return this.store.read<MemoryEntry[]>('memory.json', [])
  }

  async add(fact: string, source: string): Promise<MemoryEntry> {
    const entries = await this.list()
    const entry: MemoryEntry = {
      id: `mem_${uuidv4()}`,
      fact,
      source,
      active: true,
      createdAt: new Date().toISOString(),
    }
    entries.push(entry)
    await this.store.write('memory.json', entries)
    return entry
  }

  async update(id: string, updates: Partial<Pick<MemoryEntry, 'fact' | 'active'>>): Promise<MemoryEntry | null> {
    const entries = await this.list()
    const index = entries.findIndex((e) => e.id === id)
    if (index === -1) return null
    entries[index] = { ...entries[index], ...updates }
    await this.store.write('memory.json', entries)
    return entries[index]
  }

  async delete(id: string): Promise<void> {
    const entries = await this.list()
    const filtered = entries.filter((e) => e.id !== id)
    await this.store.write('memory.json', filtered)
  }

  parseExtractedFacts(raw: string): string[] {
    try {
      const parsed = JSON.parse(raw)
      if (Array.isArray(parsed)) {
        return parsed.filter((item): item is string => typeof item === 'string')
      }
    } catch {
      // Try regex fallback
      const match = raw.match(/\[[\s\S]*\]/)
      if (match) {
        try {
          const parsed = JSON.parse(match[0])
          if (Array.isArray(parsed)) {
            return parsed.filter((item): item is string => typeof item === 'string')
          }
        } catch {
          // Regex match also failed to parse
        }
      }
    }
    return []
  }

  async extractAndSave(userMessage: string, assistantMessage: string, source: string, ollamaChatFn: (messages: { role: string; content: string }[]) => Promise<string>): Promise<string[]> {
    const prompt = `Given this conversation exchange, extract any facts worth remembering about the user, their preferences, projects, or context. Return as a JSON array of strings. If nothing worth remembering, return empty array.

Exchange:
User: ${userMessage}
Assistant: ${assistantMessage}`

    try {
      const raw = await ollamaChatFn([{ role: 'user', content: prompt }])
      const facts = this.parseExtractedFacts(raw)
      for (const fact of facts) {
        await this.add(fact, source)
      }
      return facts
    } catch {
      return []
    }
  }
}
```

```typescript
import { v4 as uuidv4 } from 'uuid'
import type { SkillEntry, SkillDefinition, BuiltinHandler, ApiHandler, PromptHandler } from '../storage/types.ts'
import type { FileStore } from '../storage/fileStore.ts'

const BUILTIN_SKILLS: SkillEntry[] = [
  {
    id: 'skill_builtin_calculator',
    name: 'calculator',
    definition: {
      type: 'function',
      function: {
        name: 'calculator',
        description: 'Evaluate a mathematical expression and return the result',
        parameters: {
          type: 'object',
          properties: {
            expression: { type: 'string', description: "The math expression to evaluate, e.g. '2 + 2 * 3'" },
          },
          required: ['expression'],
        },
      },
    },
    handler: { type: 'builtin', action: 'calculator' },
    active: true,
    createdAt: new Date().toISOString(),
  },
  {
    id: 'skill_builtin_weather',
    name: 'get_weather',
    definition: {
      type: 'function',
      function: {
        name: 'get_weather',
        description: 'Get the current weather for a given city',
        parameters: {
          type: 'object',
          properties: {
            city: { type: 'string', description: "The city name, e.g. 'London'" },
            unit: { type: 'string', enum: ['celsius', 'fahrenheit'], description: 'Temperature unit' },
          },
          required: ['city'],
        },
      },
    },
    handler: { type: 'builtin', action: 'weather' },
    active: true,
    createdAt: new Date().toISOString(),
  },
]

export class SkillsService {
  private store: FileStore

  constructor(store: FileStore) {
    this.store = store
  }

  async init(): Promise<void> {
    const existing = await this.store.read<SkillEntry[]>('skills.json')
    if (!existing) {
      await this.store.write('skills.json', BUILTIN_SKILLS)
    }
  }

  async list(): Promise<SkillEntry[]> {
    return this.store.read<SkillEntry[]>('skills.json', [])
  }

  async add(definition: SkillDefinition, handler: BuiltinHandler | ApiHandler | PromptHandler): Promise<SkillEntry> {
    const entries = await this.list()
    const entry: SkillEntry = {
      id: `skill_${uuidv4()}`,
      name: definition.function.name,
      definition,
      handler,
      active: true,
      createdAt: new Date().toISOString(),
    }
    entries.push(entry)
    await this.store.write('skills.json', entries)
    return entry
  }

  async update(id: string, updates: Partial<Pick<SkillEntry, 'definition' | 'handler' | 'active'>>): Promise<SkillEntry | null> {
    const entries = await this.list()
    const index = entries.findIndex((e) => e.id === id)
    if (index === -1) return null
    entries[index] = { ...entries[index], ...updates }
    await this.store.write('skills.json', entries)
    return entries[index]
  }

  async delete(id: string): Promise<void> {
    const entries = await this.list()
    const filtered = entries.filter((e) => e.id !== id)
    await this.store.write('skills.json', filtered)
  }

  async getActiveDefinitions(): Promise<SkillDefinition[]> {
    const entries = await this.list()
    return entries.filter((e) => e.active).map((e) => e.definition)
  }

  async findByName(name: string): Promise<SkillEntry | null> {
    const entries = await this.list()
    return entries.find((e) => e.name === name) ?? null
  }

  async executeBuiltin(action: string, args: Record<string, unknown>): Promise<string> {
    if (action === 'calculator') {
      const expr = String(args.expression ?? '')
      // Safe eval: only allow numbers, operators, parentheses, spaces
      if (!/^[\d\s+\-*/().]+$/.test(expr)) {
        return 'Error: Invalid expression'
      }
      try {
        const result = Function(`"use strict"; return (${expr})`)()
        return String(result)
      } catch {
        return 'Error: Could not evaluate expression'
      }
    }

    if (action === 'weather') {
      const city = String(args.city ?? 'Unknown')
      const unit = String(args.unit ?? 'celsius')
      const temp = unit === 'fahrenheit' ? Math.floor(Math.random() * 40 + 50) : Math.floor(Math.random() * 20 + 10)
      return JSON.stringify({ city, temperature: temp, unit, condition: 'Partly cloudy', humidity: Math.floor(Math.random() * 40 + 40) })
    }

    return `Unknown builtin action: ${action}`
  }

  async executeApi(handler: ApiHandler, args: Record<string, unknown>): Promise<string> {
    try {
      const url = new URL(handler.url)
      if (handler.method === 'GET') {
        for (const [key, value] of Object.entries(args)) {
          url.searchParams.set(key, String(value))
        }
      }
      const res = await fetch(url.toString(), {
        method: handler.method,
        headers: handler.headers,
        body: handler.method === 'POST' ? JSON.stringify(args) : undefined,
      })
      return await res.text()
    } catch (err) {
      return `Error calling API: ${err instanceof Error ? err.message : 'Unknown error'}`
    }
  }

  async executePrompt(skillName: string, args: Record<string, unknown>): Promise<string> {
    const argsStr = Object.entries(args).map(([k, v]) => `${k}: ${v}`).join(', ')
    return `[Skill: ${skillName}] I would execute this with arguments (${argsStr}), but this is a prompt-only skill with no real execution. Let me describe what I would do based on these inputs.`
  }
}
```

```typescript
import cors from 'cors'

import { FileStore } from './storage/fileStore.ts'
import { OllamaService } from './services/ollama.ts'
import { ContextService } from './services/context.ts'
import { ConversationService } from './services/conversation.ts'
import { MemoryService } from './services/memory.ts'

import { createChatRouter } from './routes/chat.ts'
import { createModelsRouter } from './routes/models.ts'

import { createSkillsRouter } from './routes/skills.ts'

const PORT = parseInt(process.env.PORT ?? '3001', 10)
const OLLAMA_URL = process.env.OLLAMA_URL ?? 'http://localhost:11434'
const DATA_DIR = join(process.cwd(), 'data')

async function start() {
  const app = express()
  app.use(cors())
  app.use(express.json())

  // Initialize storage and services
  const store = new FileStore(DATA_DIR)
  await store.init()

  // Initialize default settings if not present
  const defaultSettings = {
    defaultModel: 'llama3.2',
    contextWindowSize: 10,
    autoExtractMemory: true,
    preferences: {
      responseDetail: null,
      technicalLevel: null,
      responseFormat: null,
      domains: null,
      personality: null,
      proactiveness: null,
      languageStyle: null,
      useOfExamples: null,
    },
    preferencesSet: false,
    activeSkillIds: ['skill_builtin_calculator', 'skill_builtin_weather'],
  }
  const existingSettings = await store.read('settings.json')
  if (!existingSettings) {
    await store.write('settings.json', defaultSettings)
  }

  const ollama = new OllamaService(OLLAMA_URL)
  const contextService = new ContextService()
  const conversationService = new ConversationService(store)
  const memoryService = new MemoryService(store)
  const skillsService = new SkillsService(store)
  await skillsService.init()

  // Mount routes
  app.use('/api/chat', createChatRouter(ollama, contextService, conversationService, memoryService, skillsService, store))
  app.use('/api/models', createModelsRouter(ollama))
  app.use('/api/conversations', createConversationsRouter(conversationService))
  app.use('/api/memory', createMemoryRouter(memoryService))
  app.use('/api/skills', createSkillsRouter(skillsService))
  app.use('/api/settings', createSettingsRouter(store))

  // Health check
  app.get('/api/health', (_req, res) => {
    res.json({ status: 'ok' })
  })

  app.listen(PORT, () => {
    console.log(`A.U.R.A. backend running on http://localhost:${PORT}`)
    console.log(`Ollama endpoint: ${OLLAMA_URL}`)
  })
}

start().catch((err) => {
  console.error('Failed to start server:', err)
  process.exit(1)
})
```

Create `server/routes/chat.ts`:

```typescript
import { Router } from 'express'
import { v4 as uuidv4 } from 'uuid'
import type { OllamaService } from '../services/ollama.ts'
import type { ContextService } from '../services/context.ts'
import type { ConversationService } from '../services/conversation.ts'
import type { MemoryService } from '../services/memory.ts'
import type { SkillsService } from '../services/skills.ts'
import type { FileStore } from '../storage/fileStore.ts'
import type { Settings } from '../storage/types.ts'

export function createChatRouter(
  ollama: OllamaService,
  contextService: ContextService,
  conversationService: ConversationService,
  memoryService: MemoryService,
  skillsService: SkillsService,
  store: FileStore,
) {
  const router = Router()

  router.post('/', async (req, res) => {
    const { conversationId, message } = req.body as { conversationId: string; message: string }

    if (!conversationId || !message) {
      res.status(400).json({ error: 'conversationId and message are required' })
      return
    }

    // Set SSE headers
    res.setHeader('Content-Type', 'text/event-stream')
    res.setHeader('Cache-Control', 'no-cache')
    res.setHeader('Connection', 'keep-alive')
    res.flushHeaders()

    const sendSSE = (event: string, data: unknown) => {
      res.write(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`)
    }

    try {
      // Load conversation and settings
      const conversation = await conversationService.get(conversationId)
      if (!conversation) {
        sendSSE('error', { error: 'Conversation not found' })
        res.end()
        return
      }

      const settings = await store.read<Settings>('settings.json')
      if (!settings) {
        sendSSE('error', { error: 'Settings not found' })
        res.end()
        return
      }

      // Save user message
      const userMessage = { id: uuidv4(), role: 'user' as const, content: message, timestamp: Date.now() }
      await conversationService.addMessage(conversationId, userMessage)

      // Load memory
      const memories = await memoryService.list()

      // Assemble context
      const contextMessages = contextService.assemble(
        conversation.messages,
        message,
        settings,
        memories,
        conversation.summary,
      )

      // Load active skills for tool calling
      const tools = await skillsService.getActiveDefinitions()

      // Send to Ollama (streaming)
      const stream = await ollama.chatStream({
        model: conversation.model || settings.defaultModel,
        messages: contextMessages.map((m) => ({ role: m.role, content: m.content })),
        stream: true,
        tools: tools.length > 0 ? tools : undefined,
      })

      let fullResponse = ''
      let toolCalls: { function: { name: string; arguments: Record<string, unknown> } }[] = []
      const reader = stream.getReader()
      const decoder = new TextDecoder()

      try {
        while (true) {
          const { done, value } = await reader.read()
          if (done) break

          const text = decoder.decode(value, { stream: true })
          const lines = text.split('\n').filter((l) => l.trim())

          for (const line of lines) {
            try {
              const parsed = JSON.parse(line)
              const content = parsed.message?.content ?? ''

              if (content) {
                fullResponse += content
                sendSSE('token', { content })
              }

              if (parsed.message?.tool_calls) {
                toolCalls = parsed.message.tool_calls
              }
            } catch {
              // Skip malformed lines
            }
          }
        }
      } finally {
        reader.releaseLock()
      }

      // Handle tool calls if any
      if (toolCalls.length > 0) {
        for (const toolCall of toolCalls) {
          const skillName = toolCall.function.name
          const args = toolCall.function.arguments

          sendSSE('tool_call', { name: skillName, args })

          const skill = await skillsService.findByName(skillName)
          let toolResult: string

          if (!skill) {
            toolResult = `Unknown tool: ${skillName}`
          } else if (skill.handler.type === 'builtin') {
            toolResult = await skillsService.executeBuiltin(skill.handler.action, args)
          } else if (skill.handler.type === 'api') {
            toolResult = await skillsService.executeApi(skill.handler, args)
          } else {
            toolResult = await skillsService.executePrompt(skillName, args)
          }

          sendSSE('tool_result', { name: skillName, result: toolResult })

          // Send tool result back to Ollama for final response
          const finalMessages = [
            ...contextMessages.map((m) => ({ role: m.role, content: m.content })),
            { role: 'assistant', content: fullResponse },
            { role: 'tool', content: toolResult },
          ]

          const finalResponse = await ollama.chat({
            model: conversation.model || settings.defaultModel,
            messages: finalMessages,
            stream: false,
          })

          const finalContent = finalResponse.message.content
          if (finalContent) {
            fullResponse = finalContent
            sendSSE('token', { content: finalContent, isFinal: true })
          }
        }
      }

      // Save AI message
      const aiMessage = { id: uuidv4(), role: 'assistant' as const, content: fullResponse, timestamp: Date.now() }
      await conversationService.addMessage(conversationId, aiMessage)

      // Check if summarization needed
      const updatedConv = await conversationService.get(conversationId)
      if (updatedConv && updatedConv.messages.length > settings.contextWindowSize && !updatedConv.summary) {
        const oldMessages = updatedConv.messages.slice(0, -settings.contextWindowSize)
        if (oldMessages.length > 0) {
          const summaryResponse = await ollama.chat({
            model: conversation.model || settings.defaultModel,
            messages: [
              { role: 'user', content: `Summarize this conversation in 2-3 sentences:\n${oldMessages.map((m) => `${m.role}: ${m.content}`).join('\n')}` },
            ],
            stream: false,
          })
          await conversationService.updateSummary(conversationId, summaryResponse.message.content)
        }
      }

      // Auto-extract memory if enabled
      let extractedFacts: string[] = []
      if (settings.autoExtractMemory) {
        extractedFacts = await memoryService.extractAndSave(
          message,
          fullResponse,
          conversationId,
          async (messages) => {
            const resp = await ollama.chat({
              model: conversation.model || settings.defaultModel,
              messages,
              stream: false,
            })
            return resp.message.content
          },
        )
      }

      sendSSE('done', { messageId: aiMessage.id, extractedFacts })
      res.end()
    } catch (err) {
      sendSSE('error', { error: err instanceof Error ? err.message : 'Unknown error' })
      res.end()
    }
  })

  return router
}
```

```typescript
import { Router } from 'express'
import type { FileStore } from '../storage/fileStore.ts'
import type { Settings } from '../storage/types.ts'

export function createSettingsRouter(store: FileStore) {
  const router = Router()

  router.get('/', async (_req, res) => {
    const settings = await store.read<Settings>('settings.json')
    if (!settings) {
      res.status(404).json({ error: 'Settings not found' })
      return
    }
    res.json(settings)
  })

  router.put('/', async (req, res) => {
    const updates = req.body as Partial<Settings>
    const current = await store.read<Settings>('settings.json')
    if (!current) {
      res.status(404).json({ error: 'Settings not found' })
      return
    }
    const merged = { ...current, ...updates }
    if (updates.preferences) {
      merged.preferences = { ...current.preferences, ...updates.preferences }
      if (Object.values(updates.preferences).some((v) => v !== null)) {
        merged.preferencesSet = true
      }
    }
    await store.write('settings.json', merged)
    res.json(merged)
  })

  return router
}
```

```typescript
      const res = await fetch('/api/memory')

      }

      // Use empty list

      setLoading(false)
    }
  }, [])

    fetchMemories()
  }, [fetchMemories])

  const addMemory = useCallback(async (fact: string) => {
    const res = await fetch('/api/memory', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ fact }),
    })
    if (res.ok) {
      const entry = await res.json()
      setMemories((prev) => [...prev, entry])
      return entry
    }
    return null
  }, [])

  const updateMemory = useCallback(async (id: string, updates: { fact?: string; active?: boolean }) => {
    const res = await fetch(`/api/memory/${id}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(updates),

    if (res.ok) {
      const entry = await res.json()
      setMemories((prev) => prev.map((m) => (m.id === id ? entry : m)))
      return entry
    }
    return null
  }, [])

  const deleteMemory = useCallback(async (id: string) => {
    const res = await fetch(`/api/memory/${id}`, { method: 'DELETE' })
    if (res.ok) {
      setMemories((prev) => prev.filter((m) => m.id !== id))
    }
  }, [])

  return { memories, loading, addMemory, updateMemory, deleteMemory, fetchMemories }
}
```

```typescript
        const data = await res.json()
        setActiveConversation(data)
        setActiveConversationId(id)
      }
    } catch {
      // Load failed
    }
  }, [])

  const createConversation = useCallback(async (model?: string) => {
    try {
      const res = await fetch('/api/conversations', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ model: model || defaultModel }),
      })
      if (res.ok) {
        const conv = await res.json()
        await fetchConversations()
        await loadConversation(conv.id)
        return conv
      }
    } catch {
      // Create failed
    }
    return null
  }, [defaultModel, fetchConversations, loadConversation])

  const sendMessage = useCallback((content: string) => {
    if (!activeConversationId) return

    setIsTyping(true)

    // Add user message to UI immediately
    const userMsg: Message = {
      id: `local-${Date.now()}`,
      role: 'user',
      content,
      timestamp: Date.now(),
    }
    setActiveConversation((prev) =>
      prev ? { ...prev, messages: [...prev.messages, userMsg] } : prev
    )

    // Add placeholder AI message
    const aiMsgId = `ai-${Date.now()}`
    const aiMsg: Message = {
      id: aiMsgId,
      role: 'assistant',
      content: '',
      timestamp: Date.now(),
      isStreaming: true,
    }
    setActiveConversation((prev) =>
      prev ? { ...prev, messages: [...prev.messages, aiMsg] } : prev
    )

    // Use EventSource alternative: fetch with streaming
    const controller = new AbortController()

    fetch('/api/chat', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ conversationId: activeConversationId, message: content }),
      signal: controller.signal,
    })
      .then(async (res) => {
        if (!res.ok || !res.body) throw new Error('Chat request failed')

        const reader = res.body.getReader()
        const decoder = new TextDecoder()
        let buffer = ''

        while (true) {
          const { done, value } = await reader.read()
          if (done) break

          buffer += decoder.decode(value, { stream: true })
          const lines = buffer.split('\n')
          buffer = lines.pop() ?? ''

          for (const line of lines) {
            if (line.startsWith('event: ')) {
              const eventType = line.slice(7)
              continue
            }
            if (line.startsWith('data: ')) {
              try {
                const data = JSON.parse(line.slice(6))

                // Handle different SSE event types via the last event type
                if (data.content && data.isFinal) {
                  // Final response after tool call
                  setActiveConversation((prev) =>
                    prev
                      ? {
                          ...prev,
                          messages: prev.messages.map((m) =>
                            m.id === aiMsgId ? { ...m, content: data.content, isStreaming: false } : m
                          ),
                        }
                      : prev
                  )
                } else if (data.content) {
                  // Streaming token
                  setActiveConversation((prev) =>
                    prev
                      ? {
                          ...prev,
                          messages: prev.messages.map((m) =>
                            m.id === aiMsgId
                              ? { ...m, content: m.content + data.content }
                              : m
                          ),
                        }
                      : prev
                  )
                } else if (data.name && data.args) {
                  // Tool call
                  setActiveConversation((prev) =>
                    prev
                      ? {
                          ...prev,
                          messages: [
                            ...prev.messages,
                            {
                              id: `tool-${Date.now()}`,
                              role: 'tool' as const,
                              content: '',
                              timestamp: Date.now(),
                              toolCallName: data.name,
                              toolCallArgs: data.args,
                            },
                          ],
                        }
                      : prev
                  )
                } else if (data.name && data.result !== undefined) {
                  // Tool result
                  setActiveConversation((prev) =>
                    prev
                      ? {
                          ...prev,
                          messages: prev.messages.map((m) =>
                            m.toolCallName === data.name && !m.toolResult
                              ? { ...m, toolResult: data.result }
                              : m
                          ),
                        }
                      : prev
                  )
                } else if (data.messageId) {
                  // Done event
                  setIsTyping(false)
                  setActiveConversation((prev) =>
                    prev
                      ? {
                          ...prev,
                          messages: prev.messages.map((m) =>
                            m.id === aiMsgId ? { ...m, isStreaming: false } : m
                          ),
                        }
                      : prev
                  )
                  // Refresh conversation list for updated title
                  fetchConversations()
                }
              } catch {
                // Skip malformed data
              }
            }
          }
        }
      })
      .catch(() => {
        setIsTyping(false)
        setActiveConversation((prev) =>
          prev
            ? {
                ...prev,
                messages: prev.messages.map((m) =>
                  m.id === aiMsgId
                    ? { ...m, content: 'Error: Failed to get response from the server.', isStreaming: false }
                    : m
                ),
              }
            : prev
        )
      })
  }, [activeConversationId, fetchConversations])

  useEffect(() => {
    fetchConversations()
  }, [fetchConversations])

  // Load first conversation on mount
  useEffect(() => {
    if (conversations.length > 0 && !activeConversationId) {
      loadConversation(conversations[0].id)
    }
  }, [conversations, activeConversationId, loadConversation])

  return {
    conversations,
    activeConversation,
    activeConversationId,
    setActiveConversationId: loadConversation,
    sendMessage,
    isTyping,
    createConversation,
    fetchConversations,
  }
}
```

```typescript
import { useState, useRef, useEffect } from 'react'
import type { OllamaModel } from '../../hooks/useSettings'

interface ModelSelectorProps {
  models: OllamaModel[]
  currentModel: string
  onModelChange: (model: string) => void
}

export function ModelSelector({ models, currentModel, onModelChange }: ModelSelectorProps) {
  const [open, setOpen] = useState(false)
  const ref = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const handler = (e: MouseEvent) => {
      if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false)
    }
    document.addEventListener('mousedown', handler)
    return () => document.removeEventListener('mousedown', handler)
  }, [])

  return (
    <div className="model-selector" ref={ref}>
      <button className="model-selector__button" onClick={() => setOpen(!open)}>
        <span className="model-selector__model-name">{currentModel}</span>
        <span className="model-selector__arrow">{open ? '▲' : '▼'}</span>
      </button>
      {open && (
        <div className="model-selector__dropdown">
          {models.map((model) => (
            <button
              key={model.name}
              className={`model-selector__option ${model.name === currentModel ? 'active' : ''}`}
              onClick={() => {
                onModelChange(model.name)
                setOpen(false)
              }}
            >
              {model.name}
            </button>
          ))}
          {models.length === 0 && (
            <span className="model-selector__empty">No models available</span>
          )}
        </div>
      )}
    </div>
  )
}
```

```typescript
import { useState } from 'react'
import { motion, AnimatePresence } from 'framer-motion'
import type { UserPreferences } from '../../hooks/useSettings'

interface OnboardingModalProps {
  open: boolean
  onComplete: (preferences: UserPreferences) => void
  onSkip: () => void
}

const QUESTIONS = [
  { key: 'responseDetail' as const, label: 'How detailed should responses be?', options: [

    { value: 'intermediate', label: 'Intermediate' },
    { value: 'advanced', label: 'Advanced' },
    { value: 'expert', label: 'Expert' },
  ]},
  { key: 'responseFormat' as const, label: 'Preferred response format?', options: [
    { value: 'code-first', label: 'Code-first' },
    { value: 'explanation-first', label: 'Explanation-first' },
    { value: 'balanced', label: 'Balanced' },
  ]},
  { key: 'domains' as const, label: 'What domains do you work in?', options: [] },
  { key: 'personality' as const, label: 'How should I behave?', options: [
    { value: 'friendly', label: 'Friendly & casual' },
    { value: 'professional', label: 'Professional & formal' },
    { value: 'witty', label: 'Witty & playful' },
    { value: 'direct', label: 'Direct & no-nonsense' },
    { value: 'adaptable', label: 'Adaptable (matches your tone)' },
  ]},
  { key: 'proactiveness' as const, label: 'How proactive should I be?', options: [
    { value: 'answer-only', label: "Answer only what's asked" },
    { value: 'suggest', label: 'Suggest alternatives' },
    { value: 'proactive', label: 'Actively offer ideas' },
  ]},
  { key: 'languageStyle' as const, label: 'Language style?', options: [
    { value: 'simple', label: 'Simple & jargon-free' },
    { value: 'technical-explained', label: 'Technical with explanations' },
    { value: 'technical-raw', label: 'Technical, skip explanations' },
  ]},
  { key: 'useOfExamples' as const, label: 'How should I use examples?', options: [
    { value: 'analogies', label: 'Analogies & metaphors' },
    { value: 'code-examples', label: 'Code examples' },
    { value: 'walkthroughs', label: 'Step-by-step walkthroughs' },
    { value: 'minimal', label: 'Minimal examples' },
  ]},
]

export function OnboardingModal({ open, onComplete, onSkip }: OnboardingModalProps) {
  const [step, setStep] = useState(0)
  const [prefs, setPrefs] = useState<UserPreferences>({
    responseDetail: null, technicalLevel: null, responseFormat: null,
    domains: null, personality: null, proactiveness: null,
    languageStyle: null, useOfExamples: null,
  })

  const question = QUESTIONS[step]

  const handleSelect = (value: string) => {
    setPrefs((prev) => ({ ...prev, [question.key]: value }))
    if (step < QUESTIONS.length - 1) {
      setStep(step + 1)
    } else {
      onComplete(prefs)
    }
  }

  const handleSkip = () => {
    if (step < QUESTIONS.length - 1) {
      setStep(step + 1)
    } else {
      onComplete(prefs)
    }
  }

  const handleSkipAll = () => {
    onSkip()
  }

  return (
    <AnimatePresence>
      {open && (
        <motion.div
          className="onboarding-overlay"
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          exit={{ opacity: 0 }}
        >
          <motion.div
            className="onboarding-modal"
            initial={{ scale: 0.9, opacity: 0 }}
            animate={{ scale: 1, opacity: 1 }}
            exit={{ scale: 0.9, opacity: 0 }}
          >
            <h2 className="onboarding-modal__title">Set up A.U.R.A.</h2>
            <p className="onboarding-modal__progress">{step + 1} / {QUESTIONS.length}</p>
            <p className="onboarding-modal__question">{question.label}</p>
            {question.options.length > 0 ? (
              <div className="onboarding-modal__options">
                {question.options.map((opt) => (
                  <button
                    key={opt.value}
                    className="onboarding-modal__option"
                    onClick={() => handleSelect(opt.value)}
                  >
                    {opt.label}
                  </button>
                ))}
              </div>
            ) : (
              <div className="onboarding-modal__free-text">
                <input
                  className="onboarding-modal__input"
                  placeholder="e.g., web development, data science"
                  onChange={(e) => setPrefs((prev) => ({ ...prev, [question.key]: e.target.value || null }))}
                />
                <button className="onboarding-modal__next" onClick={handleSkip}>
                  Next
                </button>
              </div>
            )}
            <div className="onboarding-modal__actions">
              <button className="onboarding-modal__skip" onClick={handleSkip}>Skip</button>
              <button className="onboarding-modal__skip-all" onClick={handleSkipAll}>Skip All</button>
            </div>
          </motion.div>
        </motion.div>
      )}
    </AnimatePresence>
  )
}
```

```typescript
import { motion, AnimatePresence } from 'framer-motion'

interface MemoryNotificationProps {
  facts: string[]
  onDismiss: () => void
}

export function MemoryNotification({ facts, onDismiss }: MemoryNotificationProps) {
  if (facts.length === 0) return null

  return (
    <AnimatePresence>
      <motion.div
        className="memory-notification"
        initial={{ y: 50, opacity: 0 }}
        animate={{ y: 0, opacity: 1 }}
        exit={{ y: 50, opacity: 0 }}
        onClick={onDismiss}
      >
        <span className="memory-notification__icon">🧠</span>
        <span className="memory-notification__text">
          {facts.length === 1
            ? `Memory saved: "${facts[0]}"`
            : `${facts.length} memories saved`}
        </span>
      </motion.div>
    </AnimatePresence>
  )
}
```
