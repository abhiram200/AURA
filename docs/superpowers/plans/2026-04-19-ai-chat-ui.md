# AI Chat UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a stunning dark-futuristic AI chat interface with 3D particle morphing intro, rich card AI responses, ambient animations, and comprehensive micro-interactions — all with mock data, no backend.

**Architecture:** React 18 + Vite + TypeScript app. Three.js (via R3F) renders 3D particle intro and ambient background. GSAP orchestrates intro timeline and panel transitions. Framer Motion handles UI component animations (slide-in, fade, layout). CSS custom properties drive theme switching. All state is local React state via custom hooks.

**Tech Stack:** React 18, Vite, TypeScript, @react-three/fiber, @react-three/drei, three, gsap, framer-motion, react-markdown, highlight.js, @playwright/test

Create `src/hooks/useTheme.ts`:
```typescript
import { useState, useEffect, useCallback, createContext, useContext } from 'react';

export type Theme = 'dark' | 'light';
export type AnimationIntensity = 'subtle' | 'normal' | 'intense';

interface ThemeContextValue {
  theme: Theme;
  toggleTheme: () => void;
  animationIntensity: AnimationIntensity;
  setAnimationIntensity: (intensity: AnimationIntensity) => void;
  fontSize: number;
  setFontSize: (size: number) => void;
}

export const ThemeContext = createContext<ThemeContextValue>({
  theme: 'dark',
  toggleTheme: () => {},
  animationIntensity: 'normal',
  setAnimationIntensity: () => {},
  fontSize: 16,
  setFontSize: () => {},
});

export function useTheme() {
  return useContext(ThemeContext);
}

export function ThemeProvider({ children }: { children: React.ReactNode }) {
  const [theme, setTheme] = useState<Theme>(() => {
    const saved = localStorage.getItem('theme');
    return (saved === 'light' || saved === 'dark') ? saved : 'dark';
  });

  const [animationIntensity, setAnimationIntensity] = useState<AnimationIntensity>(() => {
    const saved = localStorage.getItem('animationIntensity');
    return (saved === 'subtle' || saved === 'normal' || saved === 'intense') ? saved : 'normal';
  });

  const [fontSize, setFontSize] = useState<number>(() => {
    const saved = localStorage.getItem('fontSize');
    return saved ? parseInt(saved, 10) : 16;
  });

  useEffect(() => {
    document.documentElement.className = theme;
    localStorage.setItem('theme', theme);
  }, [theme]);

  useEffect(() => {
    localStorage.setItem('animationIntensity', animationIntensity);
  }, [animationIntensity]);

  useEffect(() => {
    localStorage.setItem('fontSize', fontSize.toString());
  }, [fontSize]);

  const toggleTheme = useCallback(() => {
    setTheme((prev) => (prev === 'dark' ? 'light' : 'dark'));
  }, []);

  return (
    <ThemeContext.Provider
      value={{ theme, toggleTheme, animationIntensity, setAnimationIntensity, fontSize, setFontSize }}
    >
      {children}
    </ThemeContext.Provider>
  );
}
```

```tsx
import { motion } from 'framer-motion';
import { useRef, useState, type ButtonHTMLAttributes, type ReactNode } from 'react';

interface GlowButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
  children: ReactNode;
  variant?: 'primary' | 'secondary' | 'ghost';
}

export function GlowButton({ children, variant = 'primary', ...props }: GlowButtonProps) {
  const btnRef = useRef<HTMLButtonElement>(null);
  const [ripple, setRipple] = useState<{ x: number; y: number; id: number } | null>(null);

  const handleClick = (e: React.MouseEvent<HTMLButtonElement>) => {
    const rect = btnRef.current?.getBoundingClientRect();
    if (rect) {
      setRipple({
        x: e.clientX - rect.left,
        y: e.clientY - rect.top,
        id: Date.now(),
      });
      setTimeout(() => setRipple(null), 600);
    }
    props.onClick?.(e);
  };

  const variantClass = `glow-btn--${variant}`;

  return (
    <motion.button
      ref={btnRef}
      className={`glow-btn ${variantClass}`}
      whileHover={{ scale: 1.02 }}
      whileTap={{ scale: 0.97 }}
      onClick={handleClick}
      {...(props as any)}
    >
      {ripple && (
        <span
          key={ripple.id}
          className="glow-btn__ripple"
          style={{ left: ripple.x, top: ripple.y }}
        />
      )}
      {children}
    </motion.button>
  );
}
```

```typescript
  const addMessage = useCallback(
    (conversationId: string, message: Message) => {
      setConversations((prev) =>
        prev.map((conv) =>
          conv.id === conversationId
            ? { ...conv, messages: [...conv.messages, message] }
            : conv
        )
      );
    },
    []
  );

  const updateMessage = useCallback(
    (conversationId: string, messageId: string, updates: Partial<Message>) => {
      setConversations((prev) =>
        prev.map((conv) =>
          conv.id === conversationId
            ? {
                ...conv,
                messages: conv.messages.map((msg) =>
                  msg.id === messageId ? { ...msg, ...updates } : msg
                ),
              }
            : conv
        )
      );
    },
    []
  );

  const sendMessage = useCallback(
    (content: string) => {
      const userMessage: Message = {
        id: `${activeConversationId}-${Date.now()}`,
        role: 'user',
        content,
        timestamp: Date.now(),
      };

      addMessage(activeConversationId, userMessage);
      setIsTyping(true);

      const aiMessageId = `${activeConversationId}-ai-${Date.now()}`;
      const aiResponse = AI_RESPONSES[Math.floor(Math.random() * AI_RESPONSES.length)];

      setTimeout(() => {
        setIsTyping(false);
        addMessage(activeConversationId, {
          id: aiMessageId,
          role: 'assistant',
          content: '',
          timestamp: Date.now(),
          isStreaming: true,
        });

        let charIndex = 0;
        streamIntervalRef.current = setInterval(() => {
          charIndex += 2;
          if (charIndex >= aiResponse.length) {
            if (streamIntervalRef.current) clearInterval(streamIntervalRef.current);
            updateMessage(activeConversationId, aiMessageId, {
              content: aiResponse,
              isStreaming: false,
            });
          } else {
            updateMessage(activeConversationId, aiMessageId, {
              content: aiResponse.slice(0, charIndex),
              isStreaming: true,
            });
          }
        }, 20);
      }, 1200);
    },
    [activeConversationId, addMessage, updateMessage]
  );

  const createConversation = useCallback(() => {
    const id = Date.now().toString();
    const newConv: Conversation = {
      id,
      title: 'New Conversation',
      messages: [],
      systemPrompt: '',
      createdAt: Date.now(),
    };
    setConversations((prev) => [newConv, ...prev]);
    setActiveConversationId(id);
  }, []);

  const updateSystemPrompt = useCallback(
    (conversationId: string, prompt: string) => {
      setConversations((prev) =>
        prev.map((conv) =>
          conv.id === conversationId ? { ...conv, systemPrompt: prompt } : conv
        )
      );
    },
    []
  );

  useEffect(() => {
    return () => {
      if (streamIntervalRef.current) clearInterval(streamIntervalRef.current);
    };
  }, []);

  return {
    conversations,
    activeConversation,
    activeConversationId,
    setActiveConversationId,
    sendMessage,
    isTyping,
    createConversation,
    updateSystemPrompt,
  };
}
```

```tsx
import { useState, useRef, useEffect } from 'react';
import { motion } from 'framer-motion';

interface ChatInputProps {
  onSend: (message: string) => void;
  disabled?: boolean;
}

export function ChatInput({ onSend, disabled = false }: ChatInputProps) {
  const [value, setValue] = useState('');
  const [isFocused, setIsFocused] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const textareaRef = useRef<HTMLTextAreaElement>(null);

  useEffect(() => {
    if (textareaRef.current) {
      textareaRef.current.style.height = 'auto';
      textareaRef.current.style.height = Math.min(textareaRef.current.scrollHeight, 150) + 'px';
    }
  }, [value]);

  const handleSubmit = () => {
    const trimmed = value.trim();
    if (!trimmed || disabled) return;
    onSend(trimmed);
    setValue('');
    setIsSubmitting(true);
    setTimeout(() => setIsSubmitting(false), 600);
  };

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      handleSubmit();
    }
  };

  return (
    <motion.div
      className={`chat-input ${isFocused ? 'chat-input--focused' : ''} ${isSubmitting ? 'chat-input--submitting' : ''}`}
      animate={isSubmitting ? { scale: [1, 0.98, 1] } : {}}
      transition={{ duration: 0.3 }}
    >
      <textarea
        ref={textareaRef}
        className="chat-input__textarea"
        value={value}
        onChange={(e) => setValue(e.target.value)}
        onFocus={() => setIsFocused(true)}
        onBlur={() => setIsFocused(false)}
        onKeyDown={handleKeyDown}
        placeholder="Type your message..."
        rows={1}
        disabled={disabled}
      />
      <motion.button
        className="chat-input__send"
        onClick={handleSubmit}
        disabled={disabled || !value.trim()}
        whileHover={{ scale: 1.05 }}
        whileTap={{ scale: 0.9 }}
      >
        <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
          <line x1="22" y1="2" x2="11" y2="13" />
          <polygon points="22 2 15 22 11 13 2 9 22 2" />
        </svg>
      </motion.button>
    </motion.div>
  );
}
```

```tsx
import { useEffect, useRef, useState } from 'react';
import { motion, AnimatePresence } from 'framer-motion';

interface StreamingTextProps {
  content: string;
  isStreaming: boolean;
}

export function StreamingText({ content, isStreaming }: StreamingTextProps) {
  const containerRef = useRef<HTMLDivElement>(null);
  const [displayedWords, setDisplayedWords] = useState<string[]>([]);
  const wordIndexRef = useRef(0);

  const words = content.split(/(\s+)/);

  useEffect(() => {
    if (!isStreaming) {
      setDisplayedWords(words);
      wordIndexRef.current = words.length;
      return;
    }

    wordIndexRef.current = 0;
    setDisplayedWords([]);

    const interval = setInterval(() => {
      wordIndexRef.current += 1;
      if (wordIndexRef.current >= words.length) {
        clearInterval(interval);
        setDisplayedWords(words);
      } else {
        setDisplayedWords(words.slice(0, wordIndexRef.current));
      }
    }, 30);

    return () => clearInterval(interval);
  }, [content, isStreaming]);

  return (
    <div ref={containerRef} className="streaming-text">
      {displayedWords.map((word, i) => (
        <motion.span
          key={i}
          initial={{ opacity: 0, y: 4 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.15 }}
          style={{ display: 'inline' }}
        >
          {word}
        </motion.span>
      ))}
      {isStreaming && (
        <motion.span
          className="streaming-cursor"
          animate={{ opacity: [1, 0] }}
          transition={{ duration: 0.6, repeat: Infinity, repeatType: 'reverse' }}
        >
          ▎
        </motion.span>
      )}
    </div>
  );
}
```

```tsx
import { useRef, useMemo } from 'react';
import { useFrame } from '@react-three/fiber';
import * as THREE from 'three';

interface ParticleOrbProps {
  progress: number; // 0 to 1, drives the morph
  particleCount?: number;
}

const PARTICLE_SHADER = {
  vertex: `
    attribute float aSize;
    attribute vec3 aTargetPosition;
    attribute float aRandom;
    uniform float uProgress;
    uniform float uTime;
    varying float vAlpha;
    varying vec3 vColor;

    void main() {
      vec3 pos = mix(position, aTargetPosition, uProgress);
      float drift = sin(uTime * 0.5 + aRandom * 6.28) * 0.15 * (1.0 - uProgress);
      pos += drift;
      vec4 mvPosition = modelViewMatrix * vec4(pos, 1.0);
      gl_PointSize = aSize * (300.0 / -mvPosition.z) * (0.5 + uProgress * 0.5);
      gl_Position = projectionMatrix * mvPosition;
      vAlpha = 0.3 + uProgress * 0.7;
      float pulse = sin(uTime * 2.0 + aRandom * 3.14) * 0.5 + 0.5;
      vColor = mix(
        vec3(0.0, 0.94, 1.0),
        vec3(0.545, 0.361, 0.965),
        aRandom + pulse * 0.3
      );
    }
  `,
  fragment: `
    varying float vAlpha;
    varying vec3 vColor;

    void main() {
      float dist = length(gl_PointCoord - vec2(0.5));
      if (dist > 0.5) discard;
      float alpha = smoothstep(0.5, 0.1, dist) * vAlpha;
      gl_FragColor = vec4(vColor, alpha);
    }
  `,
};

export function ParticleOrb({ progress, particleCount = 3000 }: ParticleOrbProps) {
  const meshRef = useRef<THREE.Points>(null);

  const { positions, targetPositions, sizes, randoms } = useMemo(() => {
    const positions = new Float32Array(particleCount * 3);
    const targetPositions = new Float32Array(particleCount * 3);
    const sizes = new Float32Array(particleCount);
    const randoms = new Float32Array(particleCount);

    for (let i = 0; i < particleCount; i++) {
      const i3 = i * 3;
      // Scattered start positions
      positions[i3] = (Math.random() - 0.5) * 20;
      positions[i3 + 1] = (Math.random() - 0.5) * 20;
      positions[i3 + 2] = (Math.random() - 0.5) * 20;

      // Sphere target positions
      const theta = Math.random() * Math.PI * 2;
      const phi = Math.acos(2 * Math.random() - 1);
      const radius = 1.5 + Math.random() * 0.3;
      targetPositions[i3] = radius * Math.sin(phi) * Math.cos(theta);
      targetPositions[i3 + 1] = radius * Math.sin(phi) * Math.sin(theta);
      targetPositions[i3 + 2] = radius * Math.cos(phi);

      sizes[i] = Math.random() * 3 + 1;
      randoms[i] = Math.random();
    }
    return { positions, targetPositions, sizes, randoms };
  }, [particleCount]);

  const uniforms = useMemo(
    () => ({
      uProgress: { value: 0 },
      uTime: { value: 0 },
    }),
    []
  );

  useFrame((state) => {
    if (meshRef.current) {
      const material = meshRef.current.material as THREE.ShaderMaterial;
      material.uniforms.uProgress.value = progress;
      material.uniforms.uTime.value = state.clock.elapsedTime;
    }
  });

  return (
    <points ref={meshRef}>
      <bufferGeometry>
        <bufferAttribute
          attach="attributes-position"
          count={particleCount}
          array={positions}
          itemSize={3}
        />
        <bufferAttribute
          attach="attributes-aTargetPosition"
          count={particleCount}
          array={targetPositions}
          itemSize={3}
        />
        <bufferAttribute
          attach="attributes-aSize"
          count={particleCount}
          array={sizes}
          itemSize={1}
        />
        <bufferAttribute
          attach="attributes-aRandom"
          count={particleCount}
          array={randoms}
          itemSize={1}
        />
      </bufferGeometry>
      <shaderMaterial
        vertexShader={PARTICLE_SHADER.vertex}
        fragmentShader={PARTICLE_SHADER.fragment}
        uniforms={uniforms}
        transparent
        depthWrite={false}
        blending={THREE.AdditiveBlending}
      />
    </points>
  );
}
```

```tsx
import { useState, useEffect, useCallback } from 'react';
import { Canvas } from '@react-three/fiber';
import { motion, AnimatePresence } from 'framer-motion';
import gsap from 'gsap';
import { ParticleOrb } from './ParticleOrb';

interface IntroSequenceProps {
  onComplete: () => void;
}

export function IntroSequence({ onComplete }: IntroSequenceProps) {
  const [progress, setProgress] = useState(0);
  const [phase, setPhase] = useState<'forming' | 'pulse' | 'morph' | 'done'>('forming');
  const [showCanvas, setShowCanvas] = useState(true);

  useEffect(() => {
    const tl = gsap.timeline({
      onComplete: () => {
        setPhase('done');
        setShowCanvas(false);
        setTimeout(onComplete, 100);
      },
    });

    // Phase 1: Orb formation (0 → 1 over 2s)
    tl.to({}, {
      duration: 2,
      onUpdate: function () {
        setProgress(this.progress());
      },
      ease: 'power2.inOut',
    });

    // Phase 2: Energy pulse (hold at 1, flash)
    tl.call(() => setPhase('pulse'), null, '+=0');
    tl.to({}, { duration: 0.5 });

    // Phase 3: Morph to UI (1 → 0, particles scatter)
    tl.call(() => setPhase('morph'), null, '+=0');
    tl.to({}, {
      duration: 1,
      onUpdate: function () {
        setProgress(1 - this.progress());
      },
      ease: 'power3.in',
    });

    return () => {
      tl.kill();
    };
  }, [onComplete]);

  return (
    <AnimatePresence>
      <motion.div
        className="intro-sequence"
        initial={{ opacity: 1 }}
        exit={{ opacity: 0 }}
        transition={{ duration: 0.5 }}
      >
        {showCanvas && (
          <Canvas
            camera={{ position: [0, 0, 6], fov: 60 }}
            style={{ position: 'absolute', inset: 0 }}
            gl={{ alpha: true, antialias: true }}
          >
            <ParticleOrb progress={progress} />
          </Canvas>
        )}
        {phase === 'pulse' && (
          <motion.div
            className="intro-pulse"
            initial={{ scale: 0, opacity: 0.8 }}
            animate={{ scale: 8, opacity: 0 }}
            transition={{ duration: 0.8, ease: 'easeOut' }}
          />
        )}
      </motion.div>
    </AnimatePresence>
  );
}
```

Create `src/components/shared/ParticleBackground.tsx`:
```tsx
import { useRef, useMemo } from 'react';
import { Canvas, useFrame } from '@react-three/fiber';
import * as THREE from 'three';

function AmbientParticles({ count = 200 }: { count?: number }) {
  const meshRef = useRef<THREE.Points>(null);

  const { positions, sizes } = useMemo(() => {
    const positions = new Float32Array(count * 3);
    const sizes = new Float32Array(count);
    for (let i = 0; i < count; i++) {
      positions[i * 3] = (Math.random() - 0.5) * 30;
      positions[i * 3 + 1] = (Math.random() - 0.5) * 20;
      positions[i * 3 + 2] = (Math.random() - 0.5) * 15 - 5;
      sizes[i] = Math.random() * 2 + 0.5;
    }
    return { positions, sizes };
  }, [count]);

  useFrame((state) => {
    if (meshRef.current) {
      meshRef.current.rotation.y = state.clock.elapsedTime * 0.02;
      meshRef.current.rotation.x = Math.sin(state.clock.elapsedTime * 0.01) * 0.1;
    }
  });

  return (
    <points ref={meshRef}>
      <bufferGeometry>
        <bufferAttribute
          attach="attributes-position"
          count={count}
          array={positions}
          itemSize={3}
        />
        <bufferAttribute
          attach="attributes-size"
          count={count}
          array={sizes}
          itemSize={1}
        />
      </bufferGeometry>
      <pointsMaterial
        size={0.04}
        color="#00f0ff"
        transparent
        opacity={0.3}
        blending={THREE.AdditiveBlending}
        depthWrite={false}
        sizeAttenuation
      />
    </points>
  );
}

export function ParticleBackground() {
  return (
    <div className="particle-background">
      <Canvas
        camera={{ position: [0, 0, 8], fov: 60 }}
        gl={{ alpha: true, antialias: false }}
        style={{ position: 'absolute', inset: 0 }}
      >
        <AmbientParticles />
      </Canvas>
    </div>
  );
}
```
