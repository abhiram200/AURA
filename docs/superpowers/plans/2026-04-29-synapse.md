# SYNAPSE Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build SYNAPSE, a native code intelligence system that indexes TS/JS codebases into a dependency graph with impact analysis, 360° context, and visual graph exploration.

**Architecture:** Two Express services — `SynapseIndexer` (scans files, parses with web-tree-sitter, resolves imports, builds graph) and `SynapseQueryService` (loads graph, answers impact/context/search queries) — store data in `.synapse/` directories. REST API at `/api/synapse`. Frontend uses Sigma.js v3 + Graphology for WebGL graph rendering.

**Tech Stack:** web-tree-sitter, Express 5, Sigma.js v3, Graphology, React 19, vitest

Create `server/services/synapse/parser.ts`:

```typescript
import Parser, { type Language, type Tree } from 'web-tree-sitter'
import { join, basename } from 'path'
import { existsSync, readFileSync } from 'fs'
import { SynapseNodeKind, type SynapseNode, type SynapseEdge, SynapseEdgeType } from './types.ts'

export interface ParseResult {
  filePath: string
  symbols: SynapseNode[]
  imports: SynapseEdge[]
  calls: SynapseEdge[]
  heritage: SynapseEdge[]
  contains: SynapseEdge[]
}

const IGNORE_DIRS = new Set([
  'node_modules', 'dist', '.git', 'build', 'coverage',
  '.next', 'target', 'vendor', '__pycache__', '.terraform',
  '.synapse', 'out', '.cache', '.vscode', '.idea',
])

export class SynapseParser {
  private ready = false
  private jsLang: Language | null = null
  private tsLang: Language | null = null
  private tsxLang: Language | null = null

  async init(): Promise<void> {
    if (this.ready) return

    await Parser.init()

    this.jsLang = await this.loadGrammar('tree-sitter-javascript.wasm')
    this.tsLang = await this.loadGrammar('tree-sitter-typescript.wasm')
    this.tsxLang = await this.loadGrammar('tree-sitter-tsx.wasm')

    this.ready = true
  }

  private async loadGrammar(filename: string): Promise<Language> {
    const searchPaths = [
      join(process.cwd(), 'server', 'grammars', filename),
      join(process.cwd(), 'node_modules', 'web-tree-sitter', filename),
      join(process.cwd(), 'node_modules', filename.replace('.wasm', ''), filename),
    ]

    for (const p of searchPaths) {
      if (existsSync(p)) {
        return await Parser.Language.load(p)
      }
    }

    throw new Error(`WASM grammar not found: ${filename}. Searched: ${searchPaths.join(', ')}`)
  }

  isReady(): boolean {
    return this.ready
  }

  selectLanguage(filePath: string): Language | null {
    const ext = filePath.split('.').pop()?.toLowerCase()
    if (ext === 'ts' || ext === 'tsx') return this.tsLang ?? this.tsxLang
    if (ext === 'js' || ext === 'jsx' || ext === 'mjs' || ext === 'cjs') return this.jsLang
    return null
  }

  isParsable(filePath: string): boolean {
    return this.selectLanguage(filePath) !== null
  }

  shouldIgnore(dirName: string): boolean {
    return IGNORE_DIRS.has(dirName)
  }

  async parseFile(filePath: string, source: string): Promise<ParseResult> {
    if (!this.ready) throw new Error('Parser not initialized. Call init() first.')

    const lang = this.selectLanguage(filePath)
    if (!lang) throw new Error(`No language for file: ${filePath}`)

    const parser = new Parser()
    parser.setLanguage(lang)
    const tree = parser.parse(source)

    const symbols: SynapseNode[] = []
    const imports: SynapseEdge[] = []
    const calls: SynapseEdge[] = []
    const heritage: SynapseEdge[] = []
    const contains: SynapseEdge[] = []

    const fileId = filePath

    this.extractSymbols(lang, tree, filePath, fileId, source, symbols, contains)
    this.extractImports(lang, tree, filePath, fileId, imports)
    this.extractCalls(lang, tree, filePath, fileId, source, calls)
    this.extractHeritage(lang, tree, filePath, fileId, heritage)

    tree.delete()
    parser.delete()

    return { filePath, symbols, imports, calls, heritage, contains }
  }

  private extractSymbols(
    lang: Language, tree: Tree, filePath: string, fileId: string,
    source: string, symbols: SynapseNode[], contains: SynapseEdge[]
  ): void {
    const queries: [string, SynapseNodeKind][] = [
      [`(function_declaration name: (identifier) @name)`, SynapseNodeKind.Function],
      [`(class_declaration name: (type_identifier) @name)`, SynapseNodeKind.Class],
      [`(interface_declaration name: (type_identifier) @name)`, SynapseNodeKind.Interface],
      [`(method_definition name: (property_identifier) @name)`, SynapseNodeKind.Method],
      [`(method_signature name: (property_identifier) @name)`, SynapseNodeKind.Method],
      [`(public_field_definition name: (property_identifier) @name)`, SynapseNodeKind.Variable],
      [`(property_signature name: (property_identifier) @name)`, SynapseNodeKind.Variable],
    ]

    for (const [queryStr, kind] of queries) {
      try {
        const query = lang.query(queryStr)
        const captures = query.captures(tree.rootNode)
        for (const { node } of captures) {
          const name = node.text
          if (!name) continue
          const symbolId = `${filePath}:${name}`
          const existing = symbols.find(s => s.id === symbolId)
          if (existing) continue

          symbols.push({
            id: symbolId,
            kind,
            name,
            filePath,
            range: { startLine: node.startPosition.row + 1, endLine: node.endPosition.row + 1 },
            exported: this.isExported(node, source),
          })
          contains.push({
            source: fileId,
            target: symbolId,
            type: SynapseEdgeType.CONTAINS,
            confidence: 1.0,
          })
        }
        query.delete()
      } catch {
        // Query not supported by this grammar version — skip
      }
    }

    // Variable declarations (const/let/var with function/arrow values)
    this.extractVariables(lang, tree, filePath, fileId, source, symbols, contains)
  }

  private extractVariables(
    lang: Language, tree: Tree, filePath: string, fileId: string,
    source: string, symbols: SynapseNode[], contains: SynapseEdge[]
  ): void {
    try {
      const query = lang.query(
        `(lexical_declaration (variable_declarator name: (identifier) @name))`
      )
      const captures = query.captures(tree.rootNode)
      for (const { node } of captures) {
        const name = node.text
        if (!name) continue
        const symbolId = `${filePath}:${name}`
        if (symbols.find(s => s.id === symbolId)) continue

        symbols.push({
          id: symbolId,
          kind: SynapseNodeKind.Variable,
          name,
          filePath,
          range: { startLine: node.startPosition.row + 1, endLine: node.endPosition.row + 1 },
          exported: false,
        })
        contains.push({
          source: fileId,
          target: symbolId,
          type: SynapseEdgeType.CONTAINS,
          confidence: 1.0,
        })
      }
      query.delete()
    } catch {
      // Skip if query not supported
    }
  }

  private isExported(node: any, source: string): boolean {
    let current = node.parent
    while (current) {
      if (current.type === 'export_statement') return true
      current = current.parent
    }
    return false
  }

  private extractImports(
    lang: Language, tree: Tree, filePath: string, fileId: string,
    imports: SynapseEdge[]
  ): void {
    try {
      const query = lang.query(
        `(import_declaration source: (string (string_fragment) @source))`
      )
      const captures = query.captures(tree.rootNode)
      for (const { node } of captures) {
        const importPath = node.text
        if (!importPath) continue
        imports.push({
          source: fileId,
          target: importPath,
          type: SynapseEdgeType.IMPORTS,
          confidence: 0.9,
          metadata: { rawImportPath: importPath },
        })
      }
      query.delete()
    } catch {
      // Skip if query not supported
    }
  }

  private extractCalls(
    lang: Language, tree: Tree, filePath: string, fileId: string,
    source: string, calls: SynapseEdge[]
  ): void {
    try {
      // Direct function calls: foo()
      const query = lang.query(
        `(call_expression function: (identifier) @name)`
      )
      const captures = query.captures(tree.rootNode)
      for (const { node } of captures) {
        const callee = node.text
        if (!callee) continue
        const callerId = this.findEnclosingSymbolId(node, filePath)
        calls.push({
          source: callerId ?? fileId,
          target: `${filePath}:${callee}`,
          type: SynapseEdgeType.CALLS,
          confidence: callerId ? 0.95 : 0.5,
          metadata: { calleeName: callee },
        })
      }
      query.delete()
    } catch {
      // Skip
    }

    try {
      // Method calls: obj.method()
      const query = lang.query(
        `(call_expression function: (member_expression property: (property_identifier) @method))`
      )
      const captures = query.captures(tree.rootNode)
      for (const { node } of captures) {
        const callee = node.text
        if (!callee) continue
        const callerId = this.findEnclosingSymbolId(node, filePath)
        calls.push({
          source: callerId ?? fileId,
          target: `${filePath}:${callee}`,
          type: SynapseEdgeType.CALLS,
          confidence: callerId ? 0.7 : 0.4,
          metadata: { calleeName: callee, isMethodCall: true },
        })
      }
      query.delete()
    } catch {
      // Skip
    }
  }

  private findEnclosingSymbolId(node: any, filePath: string): string | null {
    let current = node.parent
    while (current) {
      if (['function_declaration', 'class_declaration', 'method_definition', 'arrow_function'].includes(current.type)) {
        const nameNode = current.childForFieldName('name')
        if (nameNode) {
          return `${filePath}:${nameNode.text}`
        }
      }
      current = current.parent
    }
    return null
  }

  private extractHeritage(
    lang: Language, tree: Tree, filePath: string, fileId: string,
    heritage: SynapseEdge[]
  ): void {
    // Extends
    try {
      const query = lang.query(
        `(class_heritage (extends_clause value: (type_identifier) @parent))`
      )
      const captures = query.captures(tree.rootNode)
      for (const { node } of captures) {
        const parentName = node.text
        if (!parentName) continue
        const classNode = node.parent?.parent
        const className = classNode?.childForFieldName?.('name')?.text
        const sourceId = className ? `${filePath}:${className}` : fileId
        heritage.push({
          source: sourceId,
          target: parentName,
          type: SynapseEdgeType.EXTENDS,
          confidence: 0.9,
          metadata: { parentName },
        })
      }
      query.delete()
    } catch {
      // Skip
    }

    // Implements
    try {
      const query = lang.query(
        `(class_heritage (implements_clause type: (type_identifier) @iface))`
      )
      const captures = query.captures(tree.rootNode)
      for (const { node } of captures) {
        const ifaceName = node.text
        if (!ifaceName) continue
        const classNode = node.parent?.parent
        const className = classNode?.childForFieldName?.('name')?.text
        const sourceId = className ? `${filePath}:${className}` : fileId
        heritage.push({
          source: sourceId,
          target: ifaceName,
          type: SynapseEdgeType.IMPLEMENTS,
          confidence: 0.9,
          metadata: { interfaceName: ifaceName },
        })
      }
      query.delete()
    } catch {
      // Skip
    }
  }
}
```

```typescript
import { readdir, stat, readFile, writeFile, mkdir } from 'fs/promises'
import { join, relative, extname } from 'path'
import { existsSync } from 'fs'
import { SynapseParser } from './parser.ts'
import type {
  SynapseNode, SynapseEdge, GraphData, IndexingStatus,
  SynapseMeta, IndexingPhase,
} from './types.ts'

export class SynapseIndexer {
  private projectPath: string
  private parser: SynapseParser
  private status: IndexingStatus
  private statusListeners: Array<(status: IndexingStatus) => void> = []
  private graph: GraphData

  constructor(projectPath: string) {
    this.projectPath = projectPath
    this.parser = new SynapseParser()
    this.graph = { nodes: [], edges: [] }
    this.status = {
      phase: 'idle',
      progress: 0,
      filesProcessed: 0,
      totalFiles: 0,
      nodesFound: 0,
      edgesFound: 0,
      message: 'Idle',
    }
  }

  onStatus(listener: (status: IndexingStatus) => void): () => void {
    this.statusListeners.push(listener)
    return () => {
      this.statusListeners = this.statusListeners.filter(l => l !== listener)
    }
  }

  private updateStatus(phase: IndexingPhase, progress: number, message: string): void {
    this.status = {
      ...this.status,
      phase,
      progress,
      message,
    }
    this.statusListeners.forEach(l => l(this.status))
  }

  getStatus(): IndexingStatus {
    return { ...this.status }
  }

  async scan(): Promise<string[]> {
    this.updateStatus('scanning', 0, 'Scanning files...')
    const gitignore = await this.loadGitignore()
    const files: string[] = []
    await this.walkDir(this.projectPath, files, gitignore)
    this.updateStatus('scanning', 1, `Found ${files.length} files`)
    this.status.totalFiles = files.length
    return files
  }

  private async loadGitignore(): Promise<Set<string>> {
    const gitignorePath = join(this.projectPath, '.gitignore')
    if (!existsSync(gitignorePath)) return new Set()
    const content = await readFile(gitignorePath, 'utf-8')
    const patterns = new Set<string>()
    for (const line of content.split('\n')) {
      const trimmed = line.trim()
      if (!trimmed || trimmed.startsWith('#')) continue
      patterns.add(trimmed)
    }
    return patterns
  }

  private async walkDir(dir: string, files: string[], gitignore: Set<string>): Promise<void> {
    if (!existsSync(dir)) return
    let entries
    try {
      entries = await readdir(dir, { withFileTypes: true })
    } catch {
      return
    }

    for (const entry of entries) {
      const fullPath = join(dir, entry.name)

      if (entry.isDirectory()) {
        if (this.parser.shouldIgnore(entry.name)) continue
        if (this.isGitignored(entry.name, gitignore)) continue
        await this.walkDir(fullPath, files, gitignore)
      } else if (entry.isFile()) {
        if (this.isGitignored(entry.name, gitignore)) continue
        if (this.isSourceFile(fullPath)) {
          files.push(fullPath)
        }
      }
    }
  }

  private isGitignored(name: string, gitignore: Set<string>): boolean {
    for (const pattern of gitignore) {
      if (pattern.endsWith('/')) {
        if (name === pattern.slice(0, -1)) return true
      } else if (pattern.startsWith('*.')) {
        const ext = pattern.slice(1)
        if (name.endsWith(ext)) return true
      } else if (name === pattern) {
        return true
      }
    }
    return false
  }

  private isSourceFile(filePath: string): boolean {
    const ext = extname(filePath).toLowerCase()
    return ['.ts', '.tsx', '.js', '.jsx', '.mjs', '.cjs'].includes(ext)
  }

  // Full pipeline — implemented in Task 5
  async indexProject(): Promise<GraphData> {
    this.graph = { nodes: [], edges: [] }

    // Initialize parser
    await this.parser.init()

    // Phase 1: Scan
    const files = await this.scan()
    if (files.length === 0) {
      this.updateStatus('complete', 1, 'No source files found')
      return this.graph
    }

    // Phase 2: Parse
    this.updateStatus('parsing', 0.2, 'Parsing files...')
    const allResults = []
    for (let i = 0; i < files.length; i++) {
      const filePath = files[i]
      try {
        const source = await readFile(filePath, 'utf-8')
        const relPath = relative(this.projectPath, filePath).replace(/\\/g, '/')
        const result = await this.parser.parseFile(relPath, source)

        // Add file node
        this.graph.nodes.push({
          id: relPath,
          kind: 0 as any, // SynapseNodeKind.File — set in Task 5
          name: filePath.split(/[\\/]/).pop() ?? relPath,
          filePath: relPath,
          exported: false,
        })

        allResults.push(result)
        this.status.filesProcessed = i + 1
        this.status.nodesFound = this.graph.nodes.length
        this.updateStatus('parsing', 0.2 + (0.4 * (i / files.length)), `Parsing ${relPath}`)
      } catch (err) {
        // Skip unparseable files
      }
    }

    // Add all symbols, contains edges, imports, calls, heritage
    for (const result of allResults) {
      this.graph.nodes.push(...result.symbols)
      this.graph.edges.push(...result.contains, ...result.imports, ...result.calls, ...result.heritage)
    }

    // Phase 3: Resolve
    this.updateStatus('resolving', 0.65, 'Resolving imports...')
    this.resolveImports()
    this.resolveHeritage()

    // Phase 4: Aggregate
    this.updateStatus('aggregating', 0.8, 'Building file-level dependencies...')
    this.aggregateFileDependencies()

    // Phase 5: Save
    this.updateStatus('saving', 0.9, 'Saving graph...')
    await this.save()

    this.status.edgesFound = this.graph.edges.length
    this.updateStatus('complete', 1, `Indexed ${this.graph.nodes.length} nodes, ${this.graph.edges.length} edges`)
    return this.graph
  }

  private resolveImports(): void {
    const nodeMap = new Map(this.graph.nodes.map(n => [n.id, n]))
    const nameIndex = new Map<string, SynapseNode[]>()

    // Build name index for global lookups
    for (const node of this.graph.nodes) {
      if (node.kind !== 0 as any) continue // Skip File nodes for name index — actually index all non-file
      const existing = nameIndex.get(node.name) ?? []
      existing.push(node)
      nameIndex.set(node.name, existing)
    }

    // Index all symbol nodes by name
    for (const node of this.graph.nodes) {
      const existing = nameIndex.get(node.name) ?? []
      if (!existing.includes(node)) existing.push(node)
      nameIndex.set(node.name, existing)
    }

    // Resolve import edges to actual nodes
    const resolvedImports: SynapseEdge[] = []
    for (const edge of this.graph.edges) {
      if (edge.type !== 'IMPORTS') {
        resolvedImports.push(edge)
        continue
      }

      const importPath = edge.metadata?.rawImportPath as string
      if (!importPath) {
        resolvedImports.push(edge)
        continue
      }

      // Try to resolve the import to an actual file
      const resolved = this.resolveImportPath(edge.source, importPath, nodeMap)
      if (resolved) {
        resolvedImports.push({
          ...edge,
          target: resolved,
          confidence: 0.9,
        })
      } else {
        resolvedImports.push({
          ...edge,
          confidence: 0.5,
        })
      }
    }
    this.graph.edges = resolvedImports
  }

  private resolveImportPath(sourceId: string, importPath: string, nodeMap: Map<string, SynapseNode>): string | null {
    // Skip non-relative imports (node_modules, etc.)
    if (!importPath.startsWith('.')) return null

    // Try exact relative path
    const sourceFile = sourceId.includes(':') ? sourceId.split(':').slice(0, -1).join(':') : sourceId
    const sourceDir = sourceFile.split('/').slice(0, -1).join('/')
    const candidates = [
      join(sourceDir, importPath).replace(/\\/g, '/'),
      join(sourceDir, importPath + '.ts').replace(/\\/g, '/'),
      join(sourceDir, importPath + '.tsx').replace(/\\/g, '/'),
      join(sourceDir, importPath + '.js').replace(/\\/g, '/'),
      join(sourceDir, importPath, 'index.ts').replace(/\\/g, '/'),
      join(sourceDir, importPath, 'index.tsx').replace(/\\/g, '/'),
    ]

    for (const candidate of candidates) {
      if (nodeMap.has(candidate)) return candidate
    }
    return null
  }

  private resolveHeritage(): void {
    const nameIndex = new Map<string, SynapseNode[]>()
    for (const node of this.graph.nodes) {
      const existing = nameIndex.get(node.name) ?? []
      existing.push(node)
      nameIndex.set(node.name, existing)
    }

    const resolvedHeritage: SynapseEdge[] = []
    for (const edge of this.graph.edges) {
      if (edge.type !== 'EXTENDS' && edge.type !== 'IMPLEMENTS') {
        resolvedHeritage.push(edge)
        continue
      }

      // If target is already a node ID, keep it
      if (this.graph.nodes.some(n => n.id === edge.target)) {
        resolvedHeritage.push(edge)
        continue
      }

      // Try to resolve by name
      const matches = nameIndex.get(edge.target) ?? []
      const exactMatch = matches.find(n => n.kind.name === 'Class' || n.kind.name === 'Interface')
      if (exactMatch) {
        resolvedHeritage.push({ ...edge, target: exactMatch.id, confidence: 0.9 })
      } else {
        resolvedHeritage.push(edge)
      }
    }
    this.graph.edges = resolvedHeritage
  }

  private aggregateFileDependencies(): void {
    const fileDeps = new Map<string, Set<string>>()

    for (const edge of this.graph.edges) {
      if (edge.type === 'IMPORTS' || edge.type === 'CALLS') {
        const sourceFile = edge.source.includes(':') ? edge.source.split(':').slice(0, -1).join(':') : edge.source
        const targetFile = edge.target.includes(':') ? edge.target.split(':').slice(0, -1).join(':') : edge.target

        if (sourceFile && targetFile && sourceFile !== targetFile) {
          const key = sourceFile
          if (!fileDeps.has(key)) fileDeps.set(key, new Set())
          fileDeps.get(key)!.add(targetFile)
        }
      }
    }

    for (const [source, targets] of fileDeps) {
      for (const target of targets) {
        this.graph.edges.push({
          source,
          target,
          type: 7 as any, // SynapseEdgeType.DEPENDS_ON
          confidence: 0.8,
        })
      }
    }
  }

  async save(): Promise<void> {
    const synapseDir = join(this.projectPath, '.synapse')
    if (!existsSync(synapseDir)) {
      await mkdir(synapseDir, { recursive: true })
    }

    await writeFile(
      join(synapseDir, 'graph.json'),
      JSON.stringify(this.graph, null, 2)
    )

    const meta: SynapseMeta = {
      indexedAt: new Date().toISOString(),
      lastCommit: await this.getGitHead(),
      stats: {
        totalFiles: this.graph.nodes.filter(n => n.kind === 0 as any).length,
        totalNodes: this.graph.nodes.length,
        totalEdges: this.graph.edges.length,
        nodesByKind: this.countBy(this.graph.nodes, n => n.kind),
        edgesByType: this.countBy(this.graph.edges, e => e.type),
      },
    }

    await writeFile(
      join(synapseDir, 'meta.json'),
      JSON.stringify(meta, null, 2)
    )
  }

  private async getGitHead(): Promise<string | null> {
    const gitHead = join(this.projectPath, '.git', 'HEAD')
    if (!existsSync(gitHead)) return null
    try {
      const head = await readFile(gitHead, 'utf-8')
      return head.trim()
    } catch {
      return null
    }
  }

  private countBy<T>(items: T[], keyFn: (item: T) => unknown): Record<string, number> {
    const counts: Record<string, number> = {}
    for (const item of items) {
      const key = String(keyFn(item))
      counts[key] = (counts[key] ?? 0) + 1
    }
    return counts
  }
}
```

```typescript
import { join } from 'path'
import { existsSync } from 'fs'
import type {
  GraphData, SynapseNode, SynapseEdge, ImpactResult, ImpactLevel,
  ImpactNode, NodeContext, SearchResult, SynapseMeta,
} from './types.ts'
import { SynapseEdgeType, SynapseNodeKind } from './types.ts'

export class SynapseQueryService {
  private projectPath: string
  private graph: GraphData | null = null
  private meta: SynapseMeta | null = null
  private nodeMap: Map<string, SynapseNode> = new Map()
  private adjacency: Map<string, { edge: SynapseEdge; neighbor: SynapseNode }[]> = new Map()
  private reverseAdjacency: Map<string, { edge: SynapseEdge; neighbor: SynapseNode }[]> = new Map()

  constructor(projectPath: string) {

    const metaPath = join(this.projectPath, '.synapse', 'meta.json')

    if (!existsSync(graphPath)) {
      this.graph = { nodes: [], edges: [] }

    }

    const graphContent = await readFile(graphPath, 'utf-8')

    this.buildIndices()
  }

  private buildIndices(): void {
    if (!this.graph) return

    this.nodeMap = new Map(this.graph.nodes.map(n => [n.id, n]))
    this.adjacency = new Map()
    this.reverseAdjacency = new Map()

    for (const edge of this.graph.edges) {
      const sourceNode = this.nodeMap.get(edge.source)
      const targetNode = this.nodeMap.get(edge.target)
      if (!sourceNode || !targetNode) continue

      if (!this.adjacency.has(edge.source)) this.adjacency.set(edge.source, [])
      this.adjacency.get(edge.source)!.push({ edge, neighbor: targetNode })

      if (!this.reverseAdjacency.has(edge.target)) this.reverseAdjacency.set(edge.target, [])
      this.reverseAdjacency.get(edge.target)!.push({ edge, neighbor: sourceNode })
    }
  }

  isLoaded(): boolean {
    return this.graph !== null
  }

  getGraph(): GraphData {
    return this.graph ?? { nodes: [], edges: [] }
  }

  getMeta(): SynapseMeta | null {
    return this.meta
  }

  getNode(id: string): SynapseNode | null {
    return this.nodeMap.get(id) ?? null
  }

  impact(targetId: string, direction: 'downstream' | 'upstream' = 'downstream', maxDepth: number = 3, minConfidence: number = 0.0): ImpactResult {
    const levels: ImpactLevel[] = []
    const visited = new Set<string>()
    visited.add(targetId)

    let frontier = [targetId]
    for (let depth = 1; depth <= maxDepth; depth++) {
      const nextFrontier: string[] = []
      const levelNodes: ImpactNode[] = []

      for (const nodeId of frontier) {
        const adj = direction === 'downstream'
          ? this.reverseAdjacency.get(nodeId) ?? []
          : this.adjacency.get(nodeId) ?? []

        for (const { edge, neighbor } of adj) {
          if (visited.has(neighbor.id)) continue
          if (edge.confidence < minConfidence) continue
          // Skip CONTAINS edges for impact analysis
          if (edge.type === SynapseEdgeType.CONTAINS) continue

          visited.add(neighbor.id)
          nextFrontier.push(neighbor.id)
          levelNodes.push({
            id: neighbor.id,
            name: neighbor.name,
            kind: neighbor.kind,
            filePath: neighbor.filePath,
            edgeType: edge.type,
            confidence: edge.confidence,
          })
        }
      }

      const label = depth === 1 ? 'WILL BREAK' : depth === 2 ? 'LIKELY AFFECTED' : 'MAY NEED TESTING'
      if (levelNodes.length > 0) {
        levels.push({ depth, label, nodes: levelNodes })
      }
      frontier = nextFrontier
      if (frontier.length === 0) break
    }

    return { target: targetId, direction, levels }
  }

  context(nodeId: string): NodeContext | null {
    const node = this.nodeMap.get(nodeId)
    if (!node) return null

    const incoming = (this.reverseAdjacency.get(nodeId) ?? [])
      .filter(e => e.edge.type !== SynapseEdgeType.CONTAINS)
      .map(({ edge, neighbor }) => ({ node: neighbor, edgeType: edge.type }))

    const outgoing = (this.adjacency.get(nodeId) ?? [])
      .filter(e => e.edge.type !== SynapseEdgeType.CONTAINS)
      .map(({ edge, neighbor }) => ({ node: neighbor, edgeType: edge.type }))

    const members: SynapseNode[] = []
    if (node.kind === SynapseNodeKind.Class || node.kind === SynapseNodeKind.Interface) {
      for (const { edge, neighbor } of this.adjacency.get(nodeId) ?? []) {
        if (edge.type === SynapseEdgeType.CONTAINS && neighbor.kind !== SynapseNodeKind.File) {
          members.push(neighbor)
        }
      }
    }

    const hierarchy: { node: SynapseNode; relation: string }[] = []
    for (const { edge, neighbor } of this.adjacency.get(nodeId) ?? []) {
      if (edge.type === SynapseEdgeType.EXTENDS) hierarchy.push({ node: neighbor, relation: 'extends' })
      if (edge.type === SynapseEdgeType.IMPLEMENTS) hierarchy.push({ node: neighbor, relation: 'implements' })
    }
    for (const { edge, neighbor } of this.reverseAdjacency.get(nodeId) ?? []) {
      if (edge.type === SynapseEdgeType.EXTENDS) hierarchy.push({ node: neighbor, relation: 'extended by' })
      if (edge.type === SynapseEdgeType.IMPLEMENTS) hierarchy.push({ node: neighbor, relation: 'implemented by' })
    }

    return { node, incoming, outgoing, members, hierarchy }
  }

  search(query: string, limit: number = 50): SearchResult[] {
    if (!this.graph) return []
    const q = query.toLowerCase()
    const results: SearchResult[] = []

    for (const node of this.graph.nodes) {
      const nameMatch = node.name.toLowerCase().includes(q)
      const pathMatch = node.filePath.toLowerCase().includes(q)
      if (!nameMatch && !pathMatch) continue

      const score = nameMatch ? (node.name.toLowerCase() === q ? 1.0 : 0.7) : 0.3
      results.push({ node, score })
    }

    results.sort((a, b) => b.score - a.score)
    return results.slice(0, limit)
  }

  dependents(nodeId: string, depth: number = 1): SynapseNode[] {
    const visited = new Set<string>([nodeId])
    const result: SynapseNode[] = []
    let frontier = [nodeId]

    for (let d = 0; d < depth; d++) {
      const nextFrontier: string[] = []
      for (const id of frontier) {
        for (const { edge, neighbor } of this.reverseAdjacency.get(id) ?? []) {
          if (visited.has(neighbor.id)) continue
          if (edge.type === SynapseEdgeType.CONTAINS) continue
          visited.add(neighbor.id)
          result.push(neighbor)
          nextFrontier.push(neighbor.id)
        }
      }
      frontier = nextFrontier
    }

    return result
  }

  dependencies(nodeId: string, depth: number = 1): SynapseNode[] {
    const visited = new Set<string>([nodeId])
    const result: SynapseNode[] = []
    let frontier = [nodeId]

    for (let d = 0; d < depth; d++) {
      const nextFrontier: string[] = []
      for (const id of frontier) {
        for (const { edge, neighbor } of this.adjacency.get(id) ?? []) {
          if (visited.has(neighbor.id)) continue
          if (edge.type === SynapseEdgeType.CONTAINS) continue
          visited.add(neighbor.id)
          result.push(neighbor)
          nextFrontier.push(neighbor.id)
        }
      }
      frontier = nextFrontier
    }

    return result
  }
}
```

```typescript
export function createSynapseRouter(
  projectService: ProjectService,

  queryMap: Map<string, SynapseQueryService>,

  const router = Router()

  // POST /api/synapse/index/:projectId — Start indexing
  router.post('/index/:projectId', async (req, res) => {
    const { projectId } = req.params
    const project = await projectService.get(projectId)
    if (!project) {
      res.status(404).json({ error: 'Project not found' })
      return
    }

    // Reuse or create indexer
    let indexer = indexerMap.get(projectId)
    if (!indexer) {
      indexer = new SynapseIndexer(project.workspacePath)
      indexerMap.set(projectId, indexer)
    }

    // Start indexing in background
    indexer.indexProject()
      .then((graphData) => {
        // Update query service with fresh data
        let queryService = queryMap.get(projectId)
        if (!queryService) {
          queryService = new SynapseQueryService(project.workspacePath)
          queryMap.set(projectId, queryService)
        }
        queryService.load()
      })
      .catch((err) => {
        console.error(`SYNAPSE indexing failed for ${projectId}:`, err)
      })

    res.json({ status: 'started', projectId })
  })

  // GET /api/synapse/status/:projectId — Indexing progress
  router.get('/status/:projectId', async (_req, res) => {
    const { projectId } = _req.params
    const indexer = indexerMap.get(projectId)
    if (!indexer) {
      res.json({ phase: 'idle', progress: 0, message: 'Not indexed yet' })
      return
    }
    res.json(indexer.getStatus())
  })

  // GET /api/synapse/graph/:projectId — Full graph data
  router.get('/graph/:projectId', async (_req, res) => {
    const { projectId } = _req.params
    const queryService = queryMap.get(projectId)
    if (!queryService || !queryService.isLoaded()) {
      res.status(404).json({ error: 'Graph not available. Start indexing first.' })
      return
    }
    res.json(queryService.getGraph())
  })

  // POST /api/synapse/search/:projectId — Search nodes
  router.post('/search/:projectId', async (req, res) => {
    const { projectId } = req.params
    const { query, limit } = req.body as { query: string; limit?: number }
    if (!query) {
      res.status(400).json({ error: 'Query string required' })
      return
    }

    const queryService = queryMap.get(projectId)
    if (!queryService || !queryService.isLoaded()) {
      res.status(404).json({ error: 'Graph not available' })
      return
    }

    const results = queryService.search(query, limit ?? 50)
    res.json({ results })
  })

  // POST /api/synapse/context/:projectId — 360° context
  router.post('/context/:projectId', async (req, res) => {
    const { projectId } = req.params
    const { nodeId } = req.body as { nodeId: string }
    if (!nodeId) {
      res.status(400).json({ error: 'nodeId required' })
      return
    }

    const queryService = queryMap.get(projectId)
    if (!queryService || !queryService.isLoaded()) {
      res.status(404).json({ error: 'Graph not available' })
      return
    }

    const context = queryService.context(nodeId)
    if (!context) {
      res.status(404).json({ error: 'Node not found' })
      return
    }
    res.json(context)
  })

  // POST /api/synapse/impact/:projectId — Impact analysis
  router.post('/impact/:projectId', async (req, res) => {
    const { projectId } = req.params
    const { targetId, direction, maxDepth, minConfidence } = req.body as {
      targetId: string
      direction?: 'downstream' | 'upstream'
      maxDepth?: number
      minConfidence?: number
    }
    if (!targetId) {
      res.status(400).json({ error: 'targetId required' })
      return
    }

    const queryService = queryMap.get(projectId)
    if (!queryService || !queryService.isLoaded()) {
      res.status(404).json({ error: 'Graph not available' })
      return
    }

    const result = queryService.impact(
      targetId,
      direction ?? 'downstream',
      maxDepth ?? 3,
      minConfidence ?? 0.0,
    )
    res.json(result)
  })

  return router
}
```

```typescript
export function useSynapse(projectId: string | null) {
  const [graphData, setGraphData] = useState<SynapseGraphData | null>(null)
  const [indexingStatus, setIndexingStatus] = useState<SynapseIndexingStatus | null>(null)
  const [selectedNode, setSelectedNode] = useState<SynapseNodeData | null>(null)
  const [nodeContext, setNodeContext] = useState<SynapseNodeContext | null>(null)
  const [impactResult, setImpactResult] = useState<SynapseImpactResult | null>(null)
  const [searchResults, setSearchResults] = useState<SynapseSearchResult[]>([])
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const pollRef = useRef<ReturnType<typeof setInterval> | null>(null)

  const fetchStatus = useCallback(async () => {
    if (!projectId) return
    try {
      const res = await fetch(`/api/synapse/status/${projectId}`)
      if (res.ok) {
        const data = await res.json()
        setIndexingStatus(data)
        return data
      }
    } catch {}
    return null
  }, [projectId])

  const fetchGraph = useCallback(async () => {
    if (!projectId) return
    setLoading(true)
    setError(null)
    try {
      const res = await fetch(`/api/synapse/graph/${projectId}`)
      if (res.ok) {
        const data = await res.json()
        setGraphData(data)
      } else if (res.status === 404) {
        setGraphData(null)
      } else {
        setError('Failed to load graph')
      }
    } catch {
      setError('Network error loading graph')
    } finally {
      setLoading(false)
    }
  }, [projectId])

    if (!projectId) return
    setLoading(true)
    setError(null)
    try {
      const res = await fetch(`/api/synapse/index/${projectId}`, { method: 'POST' })
      if (!res.ok) {
        setError('Failed to start indexing')
        return
      }
      // Start polling for status
      if (pollRef.current) clearInterval(pollRef.current)
      pollRef.current = setInterval(async () => {
        const status = await fetchStatus()
        if (status && (status.phase === 'complete' || status.phase === 'error')) {
          if (pollRef.current) clearInterval(pollRef.current)
          pollRef.current = null
          if (status.phase === 'complete') {
            fetchGraph()
          } else {
            setError(status.error ?? 'Indexing failed')
          }
          setLoading(false)
        }
      }, 1000)
    } catch {
      setError('Network error starting indexing')
      setLoading(false)
    }
  }, [projectId, fetchStatus, fetchGraph])

  const search = useCallback(async (query: string) => {
    if (!projectId || !query) {

      return

        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ query }),
      })
      if (res.ok) {
        const data = await res.json()
        setSearchResults(data.results ?? [])
      }
    } catch {}
  }, [projectId])

  const fetchContext = useCallback(async (nodeId: string) => {
    if (!projectId) return
    try {
      const res = await fetch(`/api/synapse/context/${projectId}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ nodeId }),
      })
      if (res.ok) {
        const data = await res.json()
        setNodeContext(data)
      }
    } catch {}
  }, [projectId])

  const fetchImpact = useCallback(async (targetId: string, direction: 'downstream' | 'upstream' = 'downstream') => {
    if (!projectId) return
    try {
      const res = await fetch(`/api/synapse/impact/${projectId}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ targetId, direction }),
      })
      if (res.ok) {
        const data = await res.json()
        setImpactResult(data)
      }
    } catch {}
  }, [projectId])

  const selectNode = useCallback((node: SynapseNodeData | null) => {
    setSelectedNode(node)
    if (node) {
      fetchContext(node.id)
      fetchImpact(node.id)
    } else {
      setNodeContext(null)
      setImpactResult(null)
    }
  }, [fetchContext, fetchImpact])

  // Auto-load graph and start indexing on mount
  useEffect(() => {
    if (!projectId) return

    const init = async () => {
      // Try loading existing graph first
      const graphRes = await fetch(`/api/synapse/graph/${projectId}`)
      if (graphRes.ok) {
        const data = await graphRes.json()
        if (data.nodes && data.nodes.length > 0) {
          setGraphData(data)
          return
        }
      }

      // No graph yet — check status and maybe start indexing
      const status = await fetchStatus()
      if (status && (status.phase === 'idle' || status.phase === 'error')) {
        startIndexing()
      } else if (status && status.phase !== 'complete') {
        // Already indexing — poll
        pollRef.current = setInterval(async () => {
          const s = await fetchStatus()
          if (s && (s.phase === 'complete' || s.phase === 'error')) {
            if (pollRef.current) clearInterval(pollRef.current)
            pollRef.current = null
            if (s.phase === 'complete') fetchGraph()
            setLoading(false)
          }
        }, 1000)
      }
    }

    init()

    return () => {
      if (pollRef.current) {
        clearInterval(pollRef.current)
        pollRef.current = null
      }
    }
  }, [projectId])

  return {
    graphData,
    indexingStatus,
    selectedNode,
    nodeContext,
    impactResult,
    searchResults,
    loading,
    error,
    startIndexing,
    search,
    selectNode,
    fetchImpact,
  }
}
```

```tsx
import { useEffect, useRef, useCallback } from 'react'
import Graph from 'graphology'
import Sigma from 'sigma'
import forceAtlas2 from 'graphology-layout-forceatlas2'
import noverlap from 'graphology-layout-noverlap'
import type { SynapseNodeData, SynapseEdgeData } from '../../hooks/useSynapse'

const NODE_COLORS: Record<string, string> = {
  File: '#5b8def',
  Folder: '#8b8b8b',
  Function: '#4ec9b0',
  Class: '#e8a855',
  Interface: '#b57edc',
  Method: '#f5d76e',
  Variable: '#f48fb1',
}

const EDGE_COLORS: Record<string, string> = {
  IMPORTS: '#5b8def',
  CALLS: '#9c6bca',
  EXTENDS: '#e8a855',
  IMPLEMENTS: '#f48fb1',
  CONTAINS: '#4ec9b0',
  DEPENDS_ON: '#6b6b6b',
  HAS_METHOD: '#f5d76e',
  HAS_PROPERTY: '#f48fb1',
}

interface SynapseGraphProps {
  nodes: SynapseNodeData[]
  edges: SynapseEdgeData[]
  selectedNodeId: string | null
  viewMode: 'file' | 'symbol'
  onNodeClick: (node: SynapseNodeData) => void
  onNodeHover: (node: SynapseNodeData | null) => void
  nodeFilters: Set<string>
  edgeFilters: Set<string>
}

export function SynapseGraph({
  nodes, edges, selectedNodeId, viewMode,
  onNodeClick, onNodeHover, nodeFilters, edgeFilters,
}: SynapseGraphProps) {
  const containerRef = useRef<HTMLDivElement>(null)
  const sigmaRef = useRef<Sigma | null>(null)
  const graphRef = useRef<Graph | null>(null)

  const buildGraph = useCallback(() => {
    if (!nodes || nodes.length === 0) return null

    const graph = new Graph()

    const isFile = (id: string) => !id.includes(':')
    const filteredNodes = viewMode === 'file'
      ? nodes.filter(n => isFile(n.id))
      : nodes

    const filteredEdges = edges.filter(e => {
      if (viewMode === 'file' && !isFile(e.source) && !isFile(e.target)) return false
      if (!edgeFilters.has(e.type)) return false
      return true
    })

    // Filter out CONTAINS edges in file view
    const displayEdges = viewMode === 'file'
      ? filteredEdges.filter(e => e.type === 'DEPENDS_ON')
      : filteredEdges

    // Add nodes
    for (const node of filteredNodes) {
      if (!nodeFilters.has(node.kind)) continue
      if (graph.hasNode(node.id)) continue

      graph.addNode(node.id, {
        label: node.name,
        x: Math.random() * 100,
        y: Math.random() * 100,
        size: 5,
        color: NODE_COLORS[node.kind] ?? '#888888',
        kind: node.kind,
        filePath: node.filePath,
        originalNode: node,
      })
    }

    // Add edges
    for (const edge of displayEdges) {
      if (!graph.hasNode(edge.source) || !graph.hasNode(edge.target)) continue
      const edgeKey = `${edge.source}-${edge.type}-${edge.target}`
      if (graph.hasEdge(edgeKey)) continue
      try {
        graph.addEdgeWithKey(edgeKey, edge.source, edge.target, {
          color: EDGE_COLORS[edge.type] ?? '#444444',
          type: edge.type === 'CONTAINS' ? 'dashed' : 'arrow',
          size: Math.max(0.5, edge.confidence * 2),
          edgeType: edge.type,
        })
      } catch {
        // Duplicate or self-edge — skip
      }
    }

    // Size nodes by degree
    for (const node of filteredNodes) {
      if (!graph.hasNode(node.id)) continue
      const degree = graph.degree(node.id)
      graph.setNodeAttribute(node.id, 'size', Math.max(5, Math.min(20, 5 + degree * 1.5)))
    }

    return graph
  }, [nodes, edges, viewMode, nodeFilters, edgeFilters])

  useEffect(() => {
    if (!containerRef.current) return

    // Kill existing renderer
    if (sigmaRef.current) {
      sigmaRef.current.kill()
      sigmaRef.current = null
    }

    const graph = buildGraph()
    if (!graph || graph.order === 0) return

    graphRef.current = graph

    // Run ForceAtlas2 layout
    try {
      forceAtlas2.assign(graph, {
        iterations: 100,
        settings: {
          gravity: 1,
          scalingRatio: 2,
          barnesHutOptimize: true,
        },
      })
    } catch {
      // Layout may fail on very small graphs
    }

    // Run Noverlap cleanup
    try {
      noverlap.assign(graph, {
        maxIterations: 50,
        settings: { ratio: 2 },
      })
    } catch {
      // Noverlap may fail on very small graphs
    }

    // Create Sigma renderer
    const renderer = new Sigma(graph, containerRef.current, {
      minCameraRatio: 0.1,
      maxCameraRatio: 10,
      renderEdgeLabels: false,
      defaultEdgeColor: '#444444',
      defaultNodeType: 'circle',
    })

    sigmaRef.current = renderer

    // Hover highlight
    renderer.on('enterNode', ({ node }) => {
      const attrs = graph.getNodeAttributes(node)
      if (attrs.originalNode) {
        onNodeHover(attrs.originalNode)
      }
      // Dim non-connected nodes
      const neighbors = new Set(graph.neighbors(node))
      neighbors.add(node)
      graph.forEachNode((n) => {
        graph.setNodeAttribute(n, 'highlighted', neighbors.has(n))
      })
    })

    renderer.on('leaveNode', () => {
      onNodeHover(null)
      graph.forEachNode((n) => {
        graph.removeNodeAttribute(n, 'highlighted')
      })
    })

    // Click handler
    renderer.on('clickNode', ({ node }) => {
      const attrs = graph.getNodeAttributes(node)
      if (attrs.originalNode) {
        onNodeClick(attrs.originalNode)
      }
    })

    // Resize camera to fit
    const camera = renderer.getCamera()
    camera.animatedReset({ duration: 300 })

    return () => {
      renderer.kill()
      sigmaRef.current = null
    }
  }, [buildGraph, onNodeClick, onNodeHover])

  // Highlight selected node
  useEffect(() => {
    const graph = graphRef.current
    if (!graph) return

    graph.forEachNode((node) => {
      const isSelected = node === selectedNodeId
      graph.mergeNodeAttributes(node, {
        highlighted: isSelected,
      })
    })
  }, [selectedNodeId])

  return (
    <div className="synapse-graph" ref={containerRef} />
  )
}
```
